#### 3. User Model

```ruby
# app/models/user.rb
class User < ApplicationRecord
  has_secure_password

  before_create :generate_otp_secret

  def otp_uri(issuer:)
    ROTP::TOTP.new(otp_secret_key, issuer: issuer).provisioning_uri(email)
  end

  def verify_otp(code)
    totp = ROTP::TOTP.new(otp_secret_key)
    totp.verify(code, drift_behind: 15, drift_ahead: 15)
  end

  def enable_2fa!
    update(otp_required_for_login: true)
  end

  def disable_2fa!
    update(otp_required_for_login: false, otp_secret_key: nil)
    generate_otp_secret
  end

  private

  def generate_otp_secret
    self.otp_secret_key = ROTP::Base32.random
  end
end
```

#### 4. Controller Setup

```ruby
# app/controllers/two_factor_settings_controller.rb
class TwoFactorSettingsController < ApplicationController
  before_action :authenticate_user!

  def show
    @qr_code = generate_qr_code
  end

  def enable
    if current_user.verify_otp(params[:otp_code])
      current_user.enable_2fa!
      redirect_to profile_path, notice: '2FA đã được kích hoạt thành công'
    else
      redirect_to two_factor_settings_path, alert: 'Mã OTP không hợp lệ'
    end
  end

  def disable
    if current_user.verify_otp(params[:otp_code])
      current_user.disable_2fa!
      redirect_to profile_path, notice: '2FA đã được tắt'
    else
      redirect_to profile_path, alert: 'Mã OTP không hợp lệ'
    end
  end

  private

  def generate_qr_code
    qrcode = RQRCode::QRCode.new(current_user.otp_uri(issuer: 'YourAppName'))
    qrcode.as_svg(
      offset: 0,
      color: '000',
      shape_rendering: 'crispEdges',
      module_size: 6,
      standalone: true
    )
  end
end
```

```ruby
# app/controllers/sessions_controller.rb
class SessionsController < ApplicationController
  def create
    user = User.find_by(email: params[:email])

    if user&.authenticate(params[:password])
      if user.otp_required_for_login?
        session[:pre_2fa_user_id] = user.id
        redirect_to two_factor_verification_path
      else
        session[:user_id] = user.id
        redirect_to root_path, notice: 'Đăng nhập thành công'
      end
    else
      flash.now[:alert] = 'Email hoặc password không đúng'
      render :new
    end
  end

  def verify_2fa
    user = User.find(session[:pre_2fa_user_id])

    if user.verify_otp(params[:otp_code])
      session.delete(:pre_2fa_user_id)
      session[:user_id] = user.id
      redirect_to root_path, notice: 'Đăng nhập thành công'
    else
      flash.now[:alert] = 'Mã OTP không hợp lệ'
      render :two_factor_verification
    end
  end

  def destroy
    session.delete(:user_id)
    redirect_to login_path, notice: 'Đã đăng xuất'
  end
end
```

#### 5. Routes

```ruby
# config/routes.rb
Rails.application.routes.draw do
  # Authentication
  get 'login', to: 'sessions#new'
  post 'login', to: 'sessions#create'
  delete 'logout', to: 'sessions#destroy'
  get 'two_factor_verification', to: 'sessions#two_factor_verification'
  post 'verify_2fa', to: 'sessions#verify_2fa'

  # 2FA Settings
  resource :two_factor_settings, only: [:show] do
    post :enable
    post :disable
  end
end
```

#### 7. Important Notes

- **ROTP**: Gem để generate và validate Time-based OTP (TOTP)
- **RQRCode**: Gem để tạo QR code cho việc scan vào Google Authenticator
- **drift_behind/drift_ahead**: Cho phép code hợp lệ trong khoảng thời gian 15 giây trước/sau để xử lý clock skew
- **otp_secret_key**: Phải được lưu trữ an toàn, không được expose ra ngoài
- **Backup codes**: Nên implement backup codes để user có thể đăng nhập khi mất access vào authenticator app
- **Rate limiting**: Nên implement rate limiting cho OTP verification để tránh brute force attacks

#### 8. Security Best Practices

```ruby
# Thêm rate limiting cho OTP verification
class SessionsController < ApplicationController
  MAX_OTP_ATTEMPTS = 5
  OTP_LOCKOUT_TIME = 15.minutes

  def verify_2fa
    if session[:otp_attempts].to_i >= MAX_OTP_ATTEMPTS
      if session[:otp_locked_at] && Time.current < session[:otp_locked_at] + OTP_LOCKOUT_TIME
        flash.now[:alert] = 'Tài khoản tạm thời bị khóa do nhập sai quá nhiều lần'
        return render :two_factor_verification
      else
        session[:otp_attempts] = 0
        session[:otp_locked_at] = nil
      end
    end

    user = User.find(session[:pre_2fa_user_id])

    if user.verify_otp(params[:otp_code])
      session.delete(:pre_2fa_user_id)
      session.delete(:otp_attempts)
      session.delete(:otp_locked_at)
      session[:user_id] = user.id
      redirect_to root_path, notice: 'Đăng nhập thành công'
    else
      session[:otp_attempts] = session[:otp_attempts].to_i + 1
      if session[:otp_attempts] >= MAX_OTP_ATTEMPTS
        session[:otp_locked_at] = Time.current
      end
      flash.now[:alert] = 'Mã OTP không hợp lệ'
      render :two_factor_verification
    end
  end
end
```
